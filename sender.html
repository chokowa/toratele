Cookie同期対応

以下のコードを「お気に入り」に登録してください。動画再生中に押すと、ログイン情報を保持したままテレビに送信します。

javascript:(function(){var ip=localStorage.getItem('tv_ip');if(!ip){ip=prompt('Fire TVのIPを入力', '192.168.0.242');if(ip)localStorage.setItem('tv_ip',ip)}if(!ip)return;var u=null;try{var es=performance.getEntriesByType('resource');for(var e of es){if(e.name.includes('.m3u8')){u=e.name;break;}}}catch(x){}if(!u){var v=document.querySelector('video');if(v&&v.currentSrc)u=v.currentSrc;}if(!u)u=location.href;var s='http://'+ip+':8080/?url='+encodeURIComponent(u)+'&cookie='+encodeURIComponent(document.cookie);var i=new Image();i.src=s;alert('送信中🐯\n'+(u.includes('.m3u8')?'[Stream] ':'[Page] ')+u.slice(0,40)+'...');})()

※PC/スマホのブラウザでお気に入りをクリックすると発動します。
